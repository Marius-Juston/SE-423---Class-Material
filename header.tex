% ============================
% minted
% ----------------------------
% In HTML mode, tex4ht has no robust support for minted: it relies on
% Pygments output that does not survive the htlatex DVI pipeline cleanly.
% We replace minted with verbatim-style fallbacks so the document still
% compiles. The PDF build keeps full minted (with -shell-escape).
% ============================
\ifSEhtml
  % WCAG-compliant contrast for inline and block code is configured in
  % se423.cfg, which make4ht is invoked with via "-c se423". \Css{...}
  % is only valid inside a tex4ht config (.cfg) preamble.

  \usepackage{fvextra}
  % minted is NOT loaded in HTML mode. We provide a verbatim stand-in so
  % \begin{minted}[opts]{lang} ... \end{minted} still compiles. We use
  % fancyvrb's VerbatimEnvironment trick so the body is read verbatim
  % rather than tokenised by LaTeX (a +b body argument does NOT preserve
  % verbatim and breaks on '#', '%', '{', etc.).
  \newenvironment{minted}[2][]{%
    \VerbatimEnvironment
    \begin{Verbatim}[fontsize=\footnotesize]%
  }{%
    \end{Verbatim}%
  }
  % Inline replacement: \mintinline[opts]{lang}{code}
  % Note: \mintinline must accept verbatim-style content as its third arg.
  \NewDocumentCommand{\mintinline}{O{} m v}{\texttt{#3}}
  % Setting commands become no-ops in HTML mode
  \providecommand{\setminted}[2][]{}
  \providecommand{\setmintedinline}[2][]{}
  % xcolor's [table] option puts a row-colour hook (\@rowc@lors) into every
  % alignment; tex4ht fails on it inside math matrices and booktabs rules
  % ("Misplaced \noalign": HW4, HW6, syllabus). No document uses \rowcolors,
  % so in HTML mode the hook does nothing.
  \makeatletter\AtBeginDocument{\def\@rowc@lors{}}\makeatother
\else
  \usepackage{minted}
\fi

% \SETableHeaders{rows}: the header rows of the next tables (e.g. {1} or {1,4}),
% tagged as column headers in the PDF. A no-op in the HTML build (no tagpdf there).
% Use it right before a tabular, inside its table float, or once after \input{header}
% when every table of the document starts with a header row. Not with tabularx:
% its trial runs make the tagging code recurse ("TeX capacity exceeded").
\providecommand{\SETableHeaders}[1]{}
%
% tagpdf must be skipped in tex4ht/HTML mode: it requires the LaTeX
% PDF resource manager activated via \DocumentMetadata, which we
% deliberately skip under \HCode. Loading tagpdf without that setup
% triggers "PDF resource management is no active" and a cascade of
% \tag_mc_begin:n already defined errors. Only enable for the PDF
% (lualatex, no-tex4ht) build path.
\ifLuaTeX
  \ifSEhtml\else
    \usepackage{tagpdf}
    \tagpdfsetup{activate-all}
    \renewcommand{\SETableHeaders}[1]{\tagpdfsetup{table/header-rows={#1}}}
    \AtBeginEnvironment{tcolorbox}{\tagpdfparaOff}
    \AtEndEnvironment{tcolorbox}{\tagpdfparaOn}
    % End any open paragraph before a figure. A [H] float placed right after
    % a list (only a `%` line between, the repo's convention) is otherwise
    % typeset inside the paragraph that continues the list, and tagpdf's
    % paragraph begin/end hooks go out of step (fatal at \end{document}).
    % \par in vertical mode is a no-op, so other figures are unaffected.
    \AddToHook{env/figure/before}{\par}
    \AddToHook{env/table/before}{\par}  % same for tables: else the caption lands outside the table (Lab2)
    % Same for a minted block that directly follows a nested list inside an
    % \item (e.g. HW3): end the list-continuation paragraph first. When the
    % block is the first thing in an \item, set the item's label on its own
    % line first: a bare \par leaves the label pending, and it was printed
    % over the next item's label (HW1 Exercise 8: "1." on top of "2.").
    \makeatletter
    \AddToHook{env/minted/before}{\if@inlabel\leavevmode\fi\par}
    \makeatother
  \fi
\fi

% ============================
% Accessibility helpers
% ----------------------------
% \AltStart{alt} ... \AltEnd opens a tagpdf Figure structure so
% screen readers / assistive tech read the description in PDF/UA-2
% conformant PDFs (lualatex + tagpdf only — a no-op elsewhere).
% In tex4ht/HTML mode it emits an aria-label wrapper so the same
% description is announced by web screen readers.
%
% Usage pairs:
%   \AltStart{Description...}
%     \includegraphics{...}                 % or any visual content
%   \AltEnd
%
% Convenience wrappers \AltImg, \AltTikz, \AltMath are defined below
% for the most common cases.
\ifSEhtml
  % The description also becomes the alt text of an image inside the block
  % (instead of tex4ht's placeholder "PIC").
  \NewDocumentCommand{\AltStart}{m}{\HCode{<span role="img" aria-label="#1">}\Configure{GraphicsAlt}{#1}}
  \NewDocumentCommand{\AltEnd}{}{\HCode{</span>}}
\else
  % accsupp's \BeginAccSupp{Alt={text}} ... \EndAccSupp{} writes a
  % real /Alt entry into the PDF marked-content sequence. This works
  % under PDF/UA-2 with tagpdf active, nests cleanly inside the
  % auto-tagged figure / subfigure / formula structures, and is the
  % recipe the LaTeX accessibility documentation recommends for
  % attaching alt text to figures and equations.
  \usepackage{accsupp}
  \NewDocumentCommand{\AltStart}{m}{%
    \BeginAccSupp{method=plain,Alt={#1}}%
  }
  \NewDocumentCommand{\AltEnd}{}{%
    \EndAccSupp{}%
  }
\fi

% \AltImg[opts]{path}{alt} — \includegraphics with an alt description
% suitable for PDF/UA-2 (Figure structure) and HTML (aria-label).
\ifSEhtml
  % HTML: the description becomes the <img> alt attribute itself (tex4ht's alt key).
  \NewDocumentCommand{\AltImg}{O{} m m}{%
    \includegraphics[#1,alt={#3}]{#2}%
  }
\else
  \NewDocumentCommand{\AltImg}{O{} m m}{%
    \AltStart{#3}\includegraphics[#1]{#2}\AltEnd
  }
\fi

% \AltMath{alt}{math} — display equation tagged as a Formula with an
% English-prose alt description. The math is typeset via \[ ... \].
\ifSEhtml
  \NewDocumentCommand{\AltMath}{m +m}{%
    \HCode{<span role="math" aria-label="#1">}\[#2\]\HCode{</span>}%
  }
\else
  \NewDocumentCommand{\AltMath}{m +m}{%
    \BeginAccSupp{method=plain,Alt={#1}}\[#2\]\EndAccSupp{}%
  }
\fi

% ============================
% Annotated photo macro
% ----------------------------
% Wraps an \includegraphics in a tikzpicture so callers can draw
% overlays (red ellipses, arrows, etc.) on normalised image coords.
% In tex4ht/HTML mode the whole tikzpicture is rendered as a single
% SVG by dvisvgm, which silently drops embedded raster images, so we
% fall back to a plain \includegraphics. The overlay annotations are
% lost in HTML — describe them in the mandatory alt argument so the
% information is preserved for assistive tech.
% Usage:
%   \Annotated[width=\linewidth]{path/to/photo.jpg}{Alt description}{%
%       \draw[red, thick] (0.225,0.825) ellipse (0.15 and 0.15);
%   }
\NewDocumentCommand{\Annotated}{O{} m m +m}{%
  \AltStart{#3}%
  \ifSEhtml
    \includegraphics[#1]{#2}%
  \else
    \begin{tikzpicture}
      \node[anchor=south west, inner sep=0] (img) {\includegraphics[#1]{#2}};
      \begin{scope}[x={(img.south east)}, y={(img.north west)}]
        #4
      \end{scope}
    \end{tikzpicture}%
  \fi
  \AltEnd
}
